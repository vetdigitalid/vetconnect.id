async function generateDanKirimPDF() { if (!activeResepBooking) return; const hewan = document.getElementById('resep-hewan').value.trim(); const diagnosis = document.getElementById('resep-diagnosis').value.trim(); const obat = document.getElementById('resep-obat').value.trim(); const edukasi = document.getElementById('resep-edukasi').value.trim(); if (!diagnosis || !obat) { alert('Mohon isi Diagnosis dan Daftar Obat terlebih dahulu!'); return; } const tglFormatted = new Date().toLocaleDateString('id-ID', { day: 'numeric', month: 'long', year: 'numeric' }); const verifyUrl = `https://vetconnect.my.id/verify.html?id=${activeResepBooking.booking_id}`; // 1. Simpan Data Resep ke Supabase if (dbSupabase) { const { error } = await dbSupabase.from('bookings').update({ resep_hewan: hewan, resep_diagnosis: diagnosis, resep_obat: obat, resep_edukasi: edukasi, resep_tanggal: tglFormatted, status: 'Selesai', is_expired: true }).eq('booking_id', activeResepBooking.booking_id); if (error) { console.error("Gagal simpan ke Supabase:", error); alert("Peringatan: Gagal menyimpan data resep ke database. Pastikan kolom tabel di Supabase sudah dibuat."); return; } } // 2. Isi Data ke Template PDF document.getElementById('pdf-booking-id').innerText = activeResepBooking.booking_id; document.getElementById('pdf-tanggal').innerText = tglFormatted; document.getElementById('pdf-pemilik').innerText = activeResepBooking.nama_pemilik; document.getElementById('pdf-hewan').innerText = hewan || activeResepBooking.info_hewan; document.getElementById('pdf-diagnosis').innerText = diagnosis; document.getElementById('pdf-obat').innerText = obat; document.getElementById('pdf-edukasi').innerText = edukasi || '-'; // 3. Render QR Code Verifikasi const qrContainer = document.getElementById('qrcode-pdf'); qrContainer.innerHTML = ''; new QRCode(qrContainer, { text: verifyUrl, width: 85, height: 85 }); // 4. Generate Berkas PDF const element = document.getElementById('pdf-template'); element.classList.remove('hidden'); const opt = { margin: 0.3, filename: `Resep_TeleVet_${activeResepBooking.booking_id}.pdf`, image: { type: 'jpeg', quality: 0.98 }, html2canvas: { scale: 2 }, jsPDF: { unit: 'in', format: 'letter', orientation: 'portrait' } }; await html2pdf().set(opt).from(element).save(); element.classList.add('hidden'); // 5. Susun Pesan WA Pengantar const pesanWA = `Halo Kak ${activeResepBooking.nama_pemilik},\n\n` + `Berikut kami lampirkan berkas *Resep Digital Resmi & Edukasi Perawatan* dari *drh. Nur Rohman Najib* (TeleVet).\n\n` + `📄 *Berkas PDF telah terunduh di HP/Komputer Dokter.* Silakan lampirkan file PDF tersebut ke pesan ini.\n\n` + `🔍 Keaslian resep ini juga dapat diverifikasi secara publik melalui link berikut:\n` + `${verifyUrl}\n\n` + `Semoga anabul lekas sehat kembali! 🙏✨`; const waUrl = `https://wa.me/${activeResepBooking.whatsapp}?text=${encodeURIComponent(pesanWA)}`; tutupModalResep(); muatSemuaPesanan(); window.open(waUrl, '_blank'); }
